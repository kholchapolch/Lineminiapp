SQL Example 1

เพิ่ม Product Badge รุ่นใหม่

ตัวอย่างนี้เพิ่มหนึ่ง Product Badge เข้า group เดิม พร้อม threshold และ condition อย่างละหนึ่ง row. แก้ค่าที่ขึ้นต้น REPLACE_ME_ ทุกจุดก่อนใช้ 

-- CASE: เพิ่ม Product Badge รุ่นใหม่
-- Run UAT ก่อน แล้วใช้ค่ารูป/URL ของ environment ให้ถูกต้อง
SET NAMES utf8mb4;
START TRANSACTION;

SET @model_code  = 'REPLACE_ME_MODEL_CODE';
SET @badge_name  = 'REPLACE_ME_BADGE_NAME';
SET @product_url = 'REPLACE_ME_SONY_PRODUCT_URL';
SET @image_url   = 'REPLACE_ME_PROD_BLOB_IMAGE_URL';
SET @group_code  = 'telephoto-super-telephoto-lens';
SET @sort_order  = 1080;
SET @new_version = 'REPLACE_ME_YYYY-MM-DD-change-id-001';

-- 1) Preflight: group ต้องมีอยู่ และ model/code ต้องยังไม่ชน
SELECT *
FROM badge_display_groups
WHERE group_code = @group_code AND badge_type = 'product';

SELECT id, badge_code, product_model_code
FROM badge_rules
WHERE badge_code = CONCAT('product-', LOWER(@model_code))
   OR UPPER(product_model_code) = UPPER(@model_code);

-- 2) Parent rule; LAST_INSERT_ID(id) ทำให้ได้ id ทั้งกรณี insert และ rerun
INSERT INTO badge_rules (
  badge_code,
  badge_name,
  badge_type,
  rule_type,
  display_category,
  display_group,
  display_group_code,
  product_model_code,
  product_url,
  description,
  sort_order,
  is_active,
  active_from,
  active_to,
  registration_start,
  registration_end
)
VALUES (
  CONCAT('product-', LOWER(@model_code)),
  @badge_name,
  'product',
  'achievement',
  'Product Badge',
  @group_code,
  @group_code,
  @model_code,
  @product_url,
  CONCAT('Own ', @model_code),
  @sort_order,
  TRUE,
  NULL, NULL, NULL, NULL
)
ON DUPLICATE KEY UPDATE
  id = LAST_INSERT_ID(id),
  badge_name = VALUES(badge_name),
  product_model_code = VALUES(product_model_code),
  product_url = VALUES(product_url),
  description = VALUES(description),
  display_group = VALUES(display_group),
  display_group_code = VALUES(display_group_code),
  sort_order = VALUES(sort_order),
  is_active = TRUE;

SET @rule_id = LAST_INSERT_ID();

-- 3) หนึ่งรุ่น = achievement target 1
INSERT INTO badge_rule_thresholds (
  badge_rule_id,
  level,
  display_name,
  required_count,
  achieved_image_url,
  locked_image_url,
  share_image_url,
  sort_order
)
VALUES (
  @rule_id,
  'achievement',
  @badge_name,
  1,
  @image_url,
  NULL,
  @image_url,
  @sort_order
)
ON DUPLICATE KEY UPDATE
  display_name = VALUES(display_name),
  required_count = VALUES(required_count),
  achieved_image_url = VALUES(achieved_image_url),
  locked_image_url = VALUES(locked_image_url),
  share_image_url = VALUES(share_image_url),
  sort_order = VALUES(sort_order);

-- 4) SKU condition
INSERT INTO badge_rule_conditions (
  badge_rule_id,
  condition_label,
  match_type,
  required_count,
  sony_skus
)
VALUES (
  @rule_id,
  CONCAT('Own ', @model_code),
  'any',
  1,
  JSON_ARRAY(@model_code)
)
ON DUPLICATE KEY UPDATE
  match_type = VALUES(match_type),
  required_count = VALUES(required_count),
  sony_skus = VALUES(sony_skus);

-- 5) Bump version เพื่อ invalidate cache
INSERT INTO app_config (`key`, `value`)
VALUES ('badge_rules_version', @new_version)
ON DUPLICATE KEY UPDATE `value` = VALUES(`value`);

-- 6) Review ก่อน COMMIT
SELECT
  br.id,
  br.badge_code,
  br.product_model_code,
  br.display_group_code,
  brt.level,
  brt.required_count,
  brt.achieved_image_url,
  brc.match_type,
  brc.required_count AS condition_cap,
  brc.sony_skus
FROM badge_rules br
JOIN badge_rule_thresholds brt ON brt.badge_rule_id = br.id
JOIN badge_rule_conditions brc ON brc.badge_rule_id = br.id
WHERE br.id = @rule_id;

-- ผลถูกต้อง: COMMIT;
-- ผลไม่ถูกต้อง: ROLLBACK;
หลัง query review ต้องเลือกเอง COMMIT; หรือ ROLLBACK;. ตัวอย่างตั้งใจไม่ auto-commit เพื่อป้องกันการ paste แล้วเขียน PROD ทันที

Integrity checks ทั้งระบบ

-- Current counts: บันทึก before/after แต่ไม่ hardcode ว่าต้องเท่าเดิม
SELECT 'badge_display_groups' AS table_name, COUNT(*) AS row_count
FROM badge_display_groups
UNION ALL
SELECT 'badge_rules', COUNT(*) FROM badge_rules
UNION ALL
SELECT 'badge_rule_thresholds', COUNT(*) FROM badge_rule_thresholds
UNION ALL
SELECT 'badge_rule_conditions', COUNT(*) FROM badge_rule_conditions;

-- Active rules ที่ไม่มี threshold: ต้องได้ 0 rows
SELECT br.id, br.badge_code
FROM badge_rules br
LEFT JOIN badge_rule_thresholds brt ON brt.badge_rule_id = br.id
WHERE br.is_active = TRUE
GROUP BY br.id, br.badge_code
HAVING COUNT(brt.id) = 0;

-- Active rules ที่ไม่มี condition: ปกติต้องได้ 0 rows สำหรับ dataset นี้
SELECT br.id, br.badge_code
FROM badge_rules br
LEFT JOIN badge_rule_conditions brc ON brc.badge_rule_id = br.id
WHERE br.is_active = TRUE
GROUP BY br.id, br.badge_code
HAVING COUNT(brc.id) = 0;

-- Group code ที่ไม่มี metadata: ต้องได้ 0 rows
SELECT DISTINCT br.display_group_code
FROM badge_rules br
LEFT JOIN badge_display_groups bdg
  ON bdg.group_code = br.display_group_code
WHERE br.display_group_code IS NOT NULL
  AND bdg.group_code IS NULL;

-- Threshold ที่สูงกว่า condition caps: ต้อง review ทุก row ที่ได้
SELECT
  br.badge_code,
  MAX(brt.required_count) AS max_threshold,
  condition_caps.total_cap
FROM badge_rules br
JOIN badge_rule_thresholds brt ON brt.badge_rule_id = br.id
JOIN (
  SELECT
    badge_rule_id,
    SUM(
      CASE
        WHEN match_type = 'all' THEN JSON_LENGTH(sony_skus)
        ELSE required_count
      END
    ) AS total_cap
  FROM badge_rule_conditions
  GROUP BY badge_rule_id
) condition_caps ON condition_caps.badge_rule_id = br.id
GROUP BY br.id, br.badge_code, condition_caps.total_cap
HAVING MAX(brt.required_count) > condition_caps.total_cap;

-- PROD image origin ผิด: ต้องได้ 0 rows
SELECT br.badge_code, brt.level, brt.achieved_image_url
FROM badge_rule_thresholds brt
JOIN badge_rules br ON br.id = brt.badge_rule_id
WHERE brt.achieved_image_url IS NULL
   OR brt.achieved_image_url NOT LIKE
      'https://mysonybadgestorageprd.blob.core.windows.net/%';

Recovery

Rollback

Rollback rule ที่เพิ่งเพิ่ม

START TRANSACTION;

SET @rule_code = 'REPLACE_ME_NEW_RULE_CODE';

-- ตรวจ child rows ก่อนลบ
SELECT
  br.id,
  br.badge_code,
  (SELECT COUNT(*) FROM badge_rule_thresholds t WHERE t.badge_rule_id = br.id)
    AS threshold_count,
  (SELECT COUNT(*) FROM badge_rule_conditions c WHERE c.badge_rule_id = br.id)
    AS condition_count
FROM badge_rules br
WHERE br.badge_code = @rule_code;

-- ลบ parent จะ cascade child; ใช้เฉพาะ rule ที่เพิ่มใน change นี้
DELETE FROM badge_rules
WHERE badge_code = @rule_code;

SELECT ROW_COUNT() AS deleted_rules; -- ต้องเป็น 1

UPDATE app_config
SET `value` = 'REPLACE_ME_YYYY-MM-DD-rollback-001'
WHERE `key` = 'badge_rules_version';

-- COMMIT; หรือ ROLLBACK;

Rollback การแก้ rule เดิม

START TRANSACTION;

SET
  brt.required_count = OLD_REQUIRED_COUNT,
